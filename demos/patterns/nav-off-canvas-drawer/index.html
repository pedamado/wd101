Hamburger Menu / Off-Canvas Drawer (Gaveta Lateral) Hamburger Menu / Off-Canvas Drawer

Ícone de três barras que oculta a navegação principal num painel deslizante fora do ecrã, poupando espaço crítico em dispositivos móveis sem recurso a JavaScript. Three-bar toggle icon hiding primary navigation inside a sliding off-screen drawer panel, saving critical mobile screen real estate zero-JS.

<script>
  (function(){
    var btn = document.getElementById('btn-toggle-drawer');
    var drawer = document.getElementById('festival-drawer');
    var backdrop = document.getElementById('drawer-backdrop');
    var closeBtn = document.getElementById('btn-close-drawer');
    function toggle(open) {
      if(drawer) drawer.classList.toggle('open', open);
      if(backdrop) backdrop.classList.toggle('open', open);
      if(btn) btn.setAttribute('aria-expanded', open ? 'true' : 'false');
    }
    if(btn) btn.onclick = function() { toggle(true); };
    if(closeBtn) closeBtn.onclick = function() { toggle(false); };
    if(backdrop) backdrop.onclick = function() { toggle(false); };
  })();
</script>
